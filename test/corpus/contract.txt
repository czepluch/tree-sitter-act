==================
Contract with payable constructor and ensures
==================

contract C
constructor() payable
    iff
        CALLVALUE > 0
    creates
        uint balance := CALLVALUE
    ensures
        post(balance) == CALLVALUE

------------------

(source_file

==================
Multiple contracts in one file
==================

contract A
constructor(uint z)
    iff
        inRange(uint256, z + 42)
    creates
        uint x := z + 42

contract B
constructor()
    iff true
    creates
        uint y := 0

------------------
